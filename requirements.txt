\# Aurora Downloader Bot v5.0 (aiogram 3.x) - Python dependencies
# Install with:  pip install -r requirements.txt

# Telegram Bot API framework (async, aiogram 3.x)
aiogram

# Multi-platform media downloader
# (YouTube, SoundCloud, TikTok, Instagram, Twitter, Pinterest)
yt-dlp

# Async web server for keep-alive / health checks (bundled with aiogram)
aiohttp

# HTTP requests (used for thumbnails, oEmbed, cover art download)
requests

# Image processing (cover art format conversion, HD handling)
Pillow

# Audio metadata tagging (ID3/MP4/FLAC cover art + artist/title)
mutagen

# Spotify public data access (NO API keys, NO login needed!)
# spotapi simulates browser requests to Spotify's public web API.
# Enables: single tracks, full albums, full playlists (with "Download All").
# If not installed, the bot falls back to the limited oEmbed endpoint.
spotapi

# FFmpeg (system binary, not a pip package) is required by yt-dlp for:
#   - merging separate video + audio streams (YouTube, Pinterest, Instagram)
#   - converting audio to MP3 (Spotify flow, SoundCloud)
# Install via your OS package manager:
#   Debian/Ubuntu:  sudo apt install ffmpeg
#   macOS (brew):   brew install ffmpeg
#   Arch:           sudo pacman -S ffmpeg
